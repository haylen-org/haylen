# The desktop part of the Native Demo plugin, a C++ library that the desktop player and the Windows and Linux apps load. The tooling builds the target named after the plugin, `native-demo`, against the installed SDK of the engine, whose target `haylen::native` gives it `haylen/platform/native/HaylenNative.h` and nlohmann/json. The confirm screen is Objective-C++ with AppKit on macOS, Win32 on Windows and Xlib on Linux.
cmake_minimum_required(VERSION 3.28...4.4)
project(native-demo LANGUAGES CXX)

find_package(haylen CONFIG REQUIRED)
find_package(Threads REQUIRED)

add_library(native-demo NativeDemo.cpp NativeDemoPicture.cpp NativeDemoPlugin.cpp NativeDemoPng.cpp NativeDemoStore.cpp)
target_link_libraries(native-demo PRIVATE haylen::native Threads::Threads)
target_compile_features(native-demo PRIVATE cxx_std_20)
if(APPLE)
  enable_language(OBJCXX)
  target_sources(native-demo PRIVATE NativeDemoScreen.mm)
  set_source_files_properties(NativeDemoScreen.mm PROPERTIES COMPILE_OPTIONS -fobjc-arc)
  target_link_libraries(native-demo PRIVATE "-framework AppKit")
else()
  target_sources(native-demo PRIVATE NativeDemoScreen.cpp)
endif()
if(NOT WIN32 AND NOT APPLE)
  find_package(X11 REQUIRED)
  target_link_libraries(native-demo PRIVATE X11::X11)
endif()
set_target_properties(native-demo PROPERTIES CXX_EXTENSIONS OFF CXX_VISIBILITY_PRESET hidden OBJCXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
