# The desktop part of the Native Demo plugin, a C library that the desktop player and the Windows and Linux apps load. The tool `haylen.py` builds the target named after the plugin, `native-demo`, as a shared library and passes `HAYLEN_INCLUDE_DIR` with `haylen/platform/native/HaylenNative.h`. The confirm screen is Objective-C with AppKit on macOS, Win32 on Windows and Xlib on Linux.
cmake_minimum_required(VERSION 3.28...4.4)
project(native-demo LANGUAGES C)

find_package(Threads REQUIRED)

add_library(native-demo NativeDemo.c)
target_include_directories(native-demo PRIVATE "${HAYLEN_INCLUDE_DIR}")
target_link_libraries(native-demo PRIVATE Threads::Threads)
if(APPLE)
  enable_language(OBJC)
  target_sources(native-demo PRIVATE NativeDemoScreen.m)
  set_source_files_properties(NativeDemoScreen.m PROPERTIES COMPILE_OPTIONS -fobjc-arc)
  target_link_libraries(native-demo PRIVATE "-framework AppKit")
else()
  target_sources(native-demo PRIVATE NativeDemoScreen.c)
endif()
if(NOT WIN32 AND NOT APPLE)
  find_package(X11 REQUIRED)
  target_link_libraries(native-demo PRIVATE m X11::X11)
endif()
set_target_properties(native-demo PROPERTIES C_STANDARD 11 C_VISIBILITY_PRESET hidden OBJC_VISIBILITY_PRESET hidden)
